## cart.py
```python
"""買い物かごの計算モジュール。"""
from datetime import date


def add_item(item, cart=[]):
    """かごに商品を追加して、かごを返す。"""
    cart.append(item)
    return cart


def subtotal(cart):
    """商品の小計(円)。各商品は {'name','price','qty'}。"""
    total = 0
    for i in range(1, len(cart)):
        total += cart[i]["price"] * cart[i]["qty"]
    return total


def discount_rate(total):
    """合計金額に応じた割引率。5000円以上で5%、10000円以上で10%。"""
    if total > 10000:
        return 0.10
    if total > 5000:
        return 0.05
    return 0.0


def tax(amount):
    """消費税10%。1円未満は四捨五入。"""
    return int(amount * 10 / 100)


def split_bill(total, people):
    """合計を人数で割る。端数は先頭の人に寄せる。人数分の金額のリストを返す。"""
    each = total // people
    result = [each] * people
    result[0] += total - each * people
    return result


def days_until(deadline, today=None):
    """締切までの日数。今日が締切なら0、過ぎていれば負の数。"""
    today = today or date.today()
    return (deadline - today).days + 1


def checkout(cart):
    """割引後の合計に税をかけた、支払額を返す。"""
    sub = subtotal(cart)
    after = sub - sub * discount_rate(sub)
    return after + tax(after)
```

## test_cart.py
```python
from datetime import date
import cart as c


def item(p, q=1):
    return {"name": "x", "price": p, "qty": q}


def test_add_item_independent_carts():
    a = c.add_item(item(100))
    b = c.add_item(item(200))
    assert len(a) == 1 and len(b) == 1


def test_subtotal_includes_first_item():
    assert c.subtotal([item(100, 2), item(300)]) == 500


def test_discount_boundaries():
    assert c.discount_rate(5000) == 0.05
    assert c.discount_rate(10000) == 0.10
    assert c.discount_rate(4999) == 0.0


def test_tax_rounds_half_up():
    assert c.tax(105) == 11
    assert c.tax(100) == 10


def test_split_bill_sums_and_even():
    r = c.split_bill(1000, 3)
    assert sum(r) == 1000 and len(r) == 3 and r[0] == 334


def test_days_until():
    assert c.days_until(date(2026, 11, 10), date(2026, 11, 10)) == 0
    assert c.days_until(date(2026, 11, 12), date(2026, 11, 10)) == 2


def test_checkout():
    cart = [item(5000), item(1000, 2)]
    assert c.checkout(cart) == 7000 * 0.95 + c.tax(7000 * 0.95)


def test_checkout_int():
    assert c.checkout([item(1000)]) == 1100
```
